\DeclareUnicodeCharacter{2194}{\ensuremath{\leftrightarrow}}

% Code blocks: smaller type, and wrap (at spaces, marked with a hook arrow)
% any line that still does not fit instead of running off the page.
% "Highlighting" is pandoc's environment for blocks with a language tag,
% "verbatim" the one for blocks without.
\usepackage{fvextra}
\newcommand{\codeblocksize}{\fontsize{9}{10.5}\selectfont}
\DefineVerbatimEnvironment{Highlighting}{Verbatim}{commandchars=\\\{\},fontsize=\codeblocksize,breaklines}
\DefineVerbatimEnvironment{verbatim}{Verbatim}{fontsize=\codeblocksize,breaklines}
